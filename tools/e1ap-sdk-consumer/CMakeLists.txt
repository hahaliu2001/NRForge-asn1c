cmake_minimum_required(VERSION 3.20)
project(E1APInstalledConsumer LANGUAGES CXX)
find_package(NRForgeE1AP 0.1.0 EXACT CONFIG REQUIRED)
add_executable(e1ap_setup_consumer e1_setup.cpp)
target_link_libraries(e1ap_setup_consumer PRIVATE NRForge::e1ap)
target_compile_features(e1ap_setup_consumer PRIVATE cxx_std_20)
target_compile_options(e1ap_setup_consumer PRIVATE -Wall -Wextra -Werror -pedantic-errors -Wconversion -Wsign-conversion)
set_target_properties(e1ap_setup_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME installed_six_outcomes COMMAND e1ap_setup_consumer)
set(NRFORGE_CONSUMER_LINKER gold CACHE STRING "Consumer linker")
target_link_options(e1ap_setup_consumer PRIVATE -fuse-ld=${NRFORGE_CONSUMER_LINKER} -Wl,-s)
