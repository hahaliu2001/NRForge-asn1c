cmake_minimum_required(VERSION 3.20)
project(NGAPInstalledConsumer LANGUAGES CXX)
find_package(NRForgeNGAP 0.1.0 EXACT CONFIG REQUIRED)
add_executable(ngap_setup_consumer ng_setup.cpp)
target_link_libraries(ngap_setup_consumer PRIVATE NRForge::ngap)
target_compile_features(ngap_setup_consumer PRIVATE cxx_std_20)
target_compile_options(ngap_setup_consumer PRIVATE -Wall -Wextra -Werror -pedantic-errors -Wconversion -Wsign-conversion)
set_target_properties(ngap_setup_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME installed_three_outcomes COMMAND ngap_setup_consumer)

set(NRFORGE_CONSUMER_LINKER gold CACHE STRING "Consumer linker")
target_link_options(ngap_setup_consumer PRIVATE -fuse-ld=${NRFORGE_CONSUMER_LINKER} -Wl,-s)
