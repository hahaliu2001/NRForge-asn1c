cmake_minimum_required(VERSION 3.20)
project(F1APInstalledConsumer LANGUAGES CXX)
find_package(NRForgeF1AP 0.1.0 EXACT CONFIG REQUIRED)
add_executable(f1ap_setup_consumer f1_setup.cpp)
target_link_libraries(f1ap_setup_consumer PRIVATE NRForge::f1ap)
target_compile_features(f1ap_setup_consumer PRIVATE cxx_std_20)
target_compile_options(f1ap_setup_consumer PRIVATE -Wall -Wextra -Werror -pedantic-errors -Wconversion -Wsign-conversion)
set_target_properties(f1ap_setup_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME installed_three_outcomes COMMAND f1ap_setup_consumer)
set(NRFORGE_CONSUMER_LINKER gold CACHE STRING "Consumer linker")
target_link_options(f1ap_setup_consumer PRIVATE -fuse-ld=${NRFORGE_CONSUMER_LINKER} -Wl,-s)
