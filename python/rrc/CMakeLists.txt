cmake_minimum_required(VERSION 3.20)
project(NRForgeRRCPython VERSION 0.1.0 LANGUAGES CXX)
if(NOT NRFORGE_PYTHON_SDK_PREFIX)
  message(FATAL_ERROR "Provide NRFORGE_PYTHON_SDK_PREFIX: sealed installed RRC C++ SDK")
endif()
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 3.0.1 EXACT CONFIG REQUIRED)
find_package(NRForgeRRC 0.1.0 EXACT CONFIG REQUIRED PATHS "${NRFORGE_PYTHON_SDK_PREFIX}/lib/cmake/NRForgeRRC" NO_DEFAULT_PATH)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(generator "${CMAKE_CURRENT_SOURCE_DIR}/bindings/generate_bindings.py")
execute_process(COMMAND "${Python_EXECUTABLE}" "${generator}"
  --sdk-prefix "${NRFORGE_PYTHON_SDK_PREFIX}" --expected-fingerprint "${NRForgeRRC_FINGERPRINT}"
  --output "${generated}" RESULT_VARIABLE status)
if(NOT status EQUAL 0)
  message(FATAL_ERROR "Installed RRC public SDK cannot be fully bound")
endif()
include("${generated}/sources.cmake")
pybind11_add_module(_native MODULE NO_EXTRAS bindings/module.cpp ${NRFORGE_RRC_PYTHON_SOURCES})
target_include_directories(_native PRIVATE bindings "${generated}")
target_link_libraries(_native PRIVATE NRForge::rrc)
target_compile_features(_native PRIVATE cxx_std_20)
set_target_properties(_native PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(_native PRIVATE -O0 -Wall -Wextra -Werror -pedantic-errors -Wconversion -Wsign-conversion)
if(UNIX AND NOT APPLE)
  target_link_options(_native PRIVATE -Wl,--exclude-libs,ALL)
endif()
add_custom_target(nrforge_rrc_public_verify ALL COMMAND "${Python_EXECUTABLE}" "${generator}"
  --sdk-prefix "${NRFORGE_PYTHON_SDK_PREFIX}" --expected-fingerprint "${NRForgeRRC_FINGERPRINT}" --output "${generated}" --verify-only VERBATIM)
add_dependencies(_native nrforge_rrc_public_verify)
add_custom_command(TARGET _native POST_BUILD COMMAND "${Python_EXECUTABLE}" -c
  "import importlib.util,sys; s=importlib.util.spec_from_file_location('_native',sys.argv[1]); m=importlib.util.module_from_spec(s); s.loader.exec_module(m); assert len(m.roots())==64; print('PASS complete RRC extension import')"
  "$<TARGET_FILE:_native>" VERBATIM)
install(CODE "execute_process(COMMAND \"${Python_EXECUTABLE}\" \"${generator}\"
  --sdk-prefix \"${NRFORGE_PYTHON_SDK_PREFIX}\" --expected-fingerprint \"${NRForgeRRC_FINGERPRINT}\" --output \"${generated}\" --verify-only RESULT_VARIABLE status)
  if(NOT status EQUAL 0)
    message(FATAL_ERROR \"RRC public SDK changed before installation\")
  endif()")
install(TARGETS _native LIBRARY DESTINATION nrforge_rrc)
install(FILES "${generated}/conversion-schema.json" DESTINATION nrforge_rrc)
